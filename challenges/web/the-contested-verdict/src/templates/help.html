Переменные и типы

Переменные по умолчанию глобальные; local делает их локальными. Типы: nil, boolean, number, string, table, function.

local x = 10
local name = "Lua"
local pi = 3.14
local ok = true
local nothing = nil

Числа и строки

local n = tonumber("42")      -- строка -> число
local s = tostring(42)        -- число -> строка
local joined = "a" .. "b"     -- конкатенация строк: "ab"
local len = #"hello"          -- длина: 5
print(string.sub("hello", 1, 3))  -- "hel"
print(string.format("%.2f", pi))  -- "3.14"

end

Функции

local function add(a, b)
  return a + b
end

print(add(2, 3))         -- 5

-- несколько возвращаемых значений
local function minmax(a, b)
  if a < b then return a, b else return b, a end
end
local lo, hi = minmax(5, 2)

Таблицы

local t = {10, 20, 30}   -- массив (индексация с 1)
print(t[1])              -- 10
print(#t)                -- 3

local m = {name = "Lua", year = 1993}
print(m.name)            -- "Lua"
print(m["year"])         -- 1993

for i, v in ipairs(t) do print(i, v) end
for k, v in pairs(m) do print(k, v) end

fetch(url [, method [, body]]) — HTTP-запрос к сервису. Разрешены только URL, начинающиеся с http://127.0.0.1.
